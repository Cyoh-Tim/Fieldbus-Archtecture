```

  -DCMAKE_C_FLAGS="-g -O1 -fno-omit-frame-pointer -fsanitize=address,undefined -fno-sanitize-recover=undefined" \

  -DBUILD_DAEMON=ON -DBUILD_API=ON -DBUILD_CTT=ON \
```
